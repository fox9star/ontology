<?xml version="1.0" encoding="UTF-8"?>
<rdf:RDF
    xmlns:rdf="http://www.w3.org/1999/02/22-rdf-syntax-ns#"
    xmlns:rdfs="http://www.w3.org/2000/01/rdf-schema#"
    xmlns:owl="http://www.w3.org/2002/07/owl#"
    xmlns:xsd="http://www.w3.org/2001/XMLSchema#"
    xmlns:ex="https://example.org/ontology/academic#"
    xml:base="https://example.org/ontology/academic">

  <owl:Ontology rdf:about="https://example.org/ontology/academic">
    <rdfs:label xml:lang="ko">대학 수강 예시 온톨로지</rdfs:label>
    <rdfs:comment xml:lang="ko">itis.md의 대학 수강 예시를 OWL 2 어휘로 표현한 모델이다. 예시에 없는 수강 기록, 과목 코드 값, 학점 값은 임의로 추가하지 않았다.</rdfs:comment>
  </owl:Ontology>

  <owl:Class rdf:about="#Student">
    <rdfs:label xml:lang="ko">학생</rdfs:label>
    <rdfs:comment xml:lang="ko">과목을 수강할 수 있는 사람의 종류.</rdfs:comment>
  </owl:Class>

  <owl:Class rdf:about="#Professor">
    <rdfs:label xml:lang="ko">교수</rdfs:label>
    <rdfs:comment xml:lang="ko">과목을 가르칠 수 있는 사람의 종류.</rdfs:comment>
  </owl:Class>

  <owl:Class rdf:about="#Course">
    <rdfs:label xml:lang="ko">과목</rdfs:label>
    <rdfs:comment xml:lang="ko">모든 과목에는 적어도 하나의 과목 코드가 있다.</rdfs:comment>
    <rdfs:subClassOf>
      <owl:Restriction>
        <owl:onProperty rdf:resource="#courseCode"/>
        <owl:minCardinality rdf:datatype="http://www.w3.org/2001/XMLSchema#nonNegativeInteger">1</owl:minCardinality>
      </owl:Restriction>
    </rdfs:subClassOf>
  </owl:Class>

  <owl:ObjectProperty rdf:about="#enrollsIn">
    <rdfs:label xml:lang="ko">수강한다</rdfs:label>
    <rdfs:domain rdf:resource="#Student"/>
    <rdfs:range rdf:resource="#Course"/>
    <owl:inverseOf rdf:resource="#hasStudent"/>
  </owl:ObjectProperty>

  <owl:ObjectProperty rdf:about="#hasStudent">
    <rdfs:label xml:lang="ko">수강생이 있다</rdfs:label>
    <rdfs:domain rdf:resource="#Course"/>
    <rdfs:range rdf:resource="#Student"/>
  </owl:ObjectProperty>

  <owl:ObjectProperty rdf:about="#teaches">
    <rdfs:label xml:lang="ko">가르친다</rdfs:label>
    <rdfs:domain rdf:resource="#Professor"/>
    <rdfs:range rdf:resource="#Course"/>
  </owl:ObjectProperty>

  <owl:DatatypeProperty rdf:about="#courseCode">
    <rdfs:label xml:lang="ko">과목 코드</rdfs:label>
    <rdfs:domain rdf:resource="#Course"/>
    <rdfs:range rdf:resource="http://www.w3.org/2001/XMLSchema#string"/>
  </owl:DatatypeProperty>

  <owl:DatatypeProperty rdf:about="#credits">
    <rdfs:label xml:lang="ko">학점</rdfs:label>
    <rdfs:domain rdf:resource="#Course"/>
    <rdfs:range rdf:resource="http://www.w3.org/2001/XMLSchema#decimal"/>
  </owl:DatatypeProperty>

  <owl:NamedIndividual rdf:about="#KimMinSu">
    <rdf:type rdf:resource="#Student"/>
    <rdfs:label xml:lang="ko">김민수</rdfs:label>
  </owl:NamedIndividual>

  <owl:NamedIndividual rdf:about="#DatabaseIntroduction">
